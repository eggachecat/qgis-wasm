# Minimal CMake build of libb2 (reference C implementation = the autotools "no SSE" branch)
cmake_minimum_required(VERSION 3.15)
project(libb2 VERSION 0.98.1 LANGUAGES C)
# config.h is normally produced by autoconf; both x86-64 and wasm32 are little endian
include(TestBigEndian)
test_big_endian(B2_BIG_ENDIAN)
if(NOT B2_BIG_ENDIAN)
  set(B2_CONFIG "#define NATIVE_LITTLE_ENDIAN 1\n")
endif()
include(CheckSymbolExists)
check_symbol_exists(explicit_bzero "string.h" HAVE_EXPLICIT_BZERO)
if(HAVE_EXPLICIT_BZERO)
  string(APPEND B2_CONFIG "#define HAVE_EXPLICIT_BZERO 1\n")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "${B2_CONFIG}")
add_library(b2 src/blake2b-ref.c src/blake2s-ref.c src/blake2bp.c src/blake2sp.c)
target_include_directories(b2 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(b2 PRIVATE SUFFIX=)
target_include_directories(b2 PRIVATE src)
set_target_properties(b2 PROPERTIES POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
install(TARGETS b2 ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES src/blake2.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(prefix "${CMAKE_INSTALL_PREFIX}")
set(exec_prefix "\${prefix}")
set(libdir "\${prefix}/${CMAKE_INSTALL_LIBDIR}")
set(includedir "\${prefix}/${CMAKE_INSTALL_INCLUDEDIR}")
set(VERSION "${PROJECT_VERSION}")
configure_file(libb2.pc.in "${CMAKE_CURRENT_BINARY_DIR}/libb2.pc" @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libb2.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
